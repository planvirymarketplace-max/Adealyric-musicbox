   133→/* ═══════════════════════════════════════════════════
   134→   CATEGORY SIDEBAR — shared across all shop pages
   135→   ═══════════════════════════════════════════════════ */
   136→
   137→function CategorySidebar({
   138→  selected,
   139→  onChange,
   140→  onNavigate,
   141→}: {
   142→  selected: ShopCategory[] | null;
   143→  onChange: (cats: ShopCategory[] | null) => void;
   144→  onNavigate?: () => void;
   145→}) {
   146→  const [openNodes, setOpenNodes] = useState<Set<string>>(new Set());
   147→
   148→  const toggle = (slug: string) => {
   149→    setOpenNodes((prev) => {
   150→      const next = new Set(prev);
   151→      if (next.has(slug)) next.delete(slug);
   152→      else next.add(slug);
   153→      return next;
   154→    });
   155→  };
   156→
   157→  function renderNode(node: CategoryNode, depth: number) {
   158→    const hasChildren = node.children && node.children.length > 0;
   159→    const isOpen = openNodes.has(node.slug);
   160→    const isLeaf = !!node.categories;
   161→
   162→    return (
   163→      <div key={node.slug}>
   164→        <button
   165→          onClick={() => {
   166→            if (isLeaf) {
   167→              onChange(node.categories!);
   168→              onNavigate?.();
   169→            } else if (hasChildren) toggle(node.slug);
   170→          }}
   171→          className={`w-full flex items-center justify-between py-1.5 text-left text-[11px] tracking-wide cursor-pointer transition-colors hover:text-black ${
   172→            isLeaf && selected && JSON.stringify(selected) === JSON.stringify(node.categories)
   173→              ? "text-black font-medium"
   174→              : "text-black/50"
   175→          }`}
   176→          style={{ paddingLeft: `${depth * 14}px` }}
   177→        >
   178→          <span>{node.label}</span>
   179→          {hasChildren && (
   180→            <ChevronDown className={`h-3 w-3 transition-transform flex-shrink-0 ${isOpen ? "rotate-180" : ""}`} />
   181→          )}
   182→        </button>
   183→        {hasChildren && isOpen && (
   184→          <div>{node.children.map((child: CategoryNode) => renderNode(child, depth + 1))}</div>
   185→        )}
   186→      </div>
   187→    );
   188→  }
   189→
   190→  return (
   191→    <div className="w-full">
   192→      <div className="flex items-center justify-between mb-4">
   193→        <p className="text-[10px] uppercase tracking-[0.2em] text-black/30">Categories</p>
   194→        {selected && (
   195→          <button onClick={() => onChange(null)} className="text-[10px] uppercase tracking-wider text-black/30 hover:text-black cursor-pointer underline">Clear</button>
   196→        )}
   197→      </div>
   198→      {CATEGORY_TREE.map((node) => renderNode(node, 0))}
   199→    </div>
   200→  );
   201→}

   207→const SIZE_LABELS: Record<string, string> = {
   208→  clothing: "Clothing Size",
   209→  waist: "Waist Size",
   210→  shoe: "Shoe Size",
   211→  hat: "Hat Size",
   212→  drinkware: "Size",
   213→  "one-size": "",
   214→  none: "",

   216→
   217→function SizeFilter({ categories, selected, onChange }: { categories: ShopCategory[] | null; selected: string | null; onChange: (size: string | null) => void }) {
   218→  let activeSizeType: SizeType = "none";
   219→  if (categories) {
   220→    const types = new Set(categories.map((c) => CATEGORY_SIZE_TYPE[c] ?? "none"));
   221→    if (types.size === 1) activeSizeType = [...types][0];
   222→    else activeSizeType = "clothing";
   223→  }
   224→
   225→  const options = SIZE_OPTIONS[activeSizeType];
   226→  const label = SIZE_LABELS[activeSizeType];
   227→  if (!options || options.length === 0 || !label) return null;
   228→
   229→  return (
   230→    <div className="mt-6">
   231→      <p className="text-[10px] uppercase tracking-[0.2em] text-black/30 mb-3">{label}</p>
   232→      <div className="flex flex-wrap gap-2">
   233→        {options.map((s) => (
   234→          <button key={s} onClick={() => onChange(selected === s ? null : s)} className={`px-3 py-1.5 text-[11px] uppercase tracking-wider border transition-all cursor-pointer ${selected === s ? "bg-black text-white border-black" : "bg-white text-black/50 border-black/15 hover:border-black/40"}`}>{s}</button>
   235→        ))}
   236→      </div>
   237→    </div>
   238→  );
   239→}

   528→export function ProductDetailPage() {
   529→  const { detailSlug, setDetailSlug, addToCart, setActiveTab, setShopSubPath } = useAppStore();
   530→  const [selectedSize, setSelectedSize] = useState<string | null>(null);
   531→  const [notifyEmail, setNotifyEmail] = useState("");
   532→  const [notifySent, setNotifySent] = useState(false);
   533→  const [sidebarCats, setSidebarCats] = useState<ShopCategory[] | null>(null);
   534→
   535→  const product = useMemo(() => ALL_PRODUCTS.find((p) => p.slug === detailSlug), [detailSlug]);
   536→  const available = product ? product.stock > 0 : false;
   537→  const enriched = product ? enrichProduct(product as ShopProduct) : null;
   538→
   539→  const related = useMemo(() => {
   540→    if (!product) return [];
   541→    return ALL_PRODUCTS.filter((p) => p.albumSlug === product.albumSlug && p.slug !== product.slug).slice(0, 4);
   542→  }, [product]);
   543→
   544→  const openRelated = useCallback((p: EnrichedProduct) => {
   545→    setDetailSlug(p.slug, "product");
   546→    setSelectedSize(null);
   547→    setNotifySent(false);
   548→  }, [setDetailSlug]);
   549→
   550→  const handleAddToCart = useCallback(() => { if (available) addToCart(); }, [available, addToCart]);
   551→  const handleNotify = useCallback(() => { if (notifyEmail) setNotifySent(true); }, [notifyEmail]);
   552→
   553→  if (!product || !enriched) {
   554→    return <div className="flex min-h-screen items-center justify-center bg-white"><p className="text-sm text-black/40">Product not found.</p></div>;
   555→  }
   556→
   557→  return (
   558→    <section className="min-h-screen bg-white px-4 pt-32 pb-20 md:px-8 md:pt-44 md:pb-28">
   559→      <div className="mx-auto max-w-[1400px]">
   560→        {/* Back + nav */}
   561→        <div className="mb-8 flex items-center gap-6">
   562→          <button
   563→            onClick={() => {
   564→              const parentAlbum = SHOP_ALBUMS.find((a) => a.slug === product.albumSlug);
   565→              if (parentAlbum) { setDetailSlug(parentAlbum.slug, "album"); }
   566→              else { setDetailSlug(null, null); setActiveTab("shop"); setShopSubPath(null); }
   567→            }}
   568→            className="flex items-center gap-2 text-[11px] uppercase tracking-[0.2em] text-black hover:text-black transition-colors cursor-pointer"
   569→          >
   570→            <ArrowLeft className="h-4 w-4" />
   571→            Back
   572→          </button>
   573→          <button onClick={() => { setDetailSlug(null, null); setActiveTab("shop"); setShopSubPath("shop-all"); }} className="text-[11px] uppercase tracking-[0.2em] text-black hover:text-black transition-colors cursor-pointer">Shop All</button>
   574→          <button onClick={() => { setDetailSlug(null, null); setActiveTab("shop"); setShopSubPath("collections"); }} className="text-[11px] uppercase tracking-[0.2em] text-black hover:text-black transition-colors cursor-pointer">Collections</button>
   575→        </div>
   576→
   577→        <div className="grid grid-cols-1 gap-8 md:grid-cols-[180px_1fr_1fr] md:gap-0">
   578→          {/* Category Sidebar */}
   579→          <div className="hidden md:block pr-8 border-r border-black/10">
   580→            <CategorySidebar selected={sidebarCats} onChange={setSidebarCats} onNavigate={() => { setDetailSlug(null, null); setActiveTab("shop"); }} />
   581→          </div>
   582→
   583→          {/* Image */}
   584→          <div className="relative aspect-square overflow-hidden bg-[#f4f4f4]">
   585→            {product.image ? (
   586→              <img src={product.image} alt={product.name} className="h-full w-full object-cover" />
   587→            ) : (
   588→              <div className="flex h-full w-full items-center justify-center"><ShoppingBag className="h-12 w-12 text-black/10" /></div>
   589→            )}
   590→            {!available && (
   591→              <div className="absolute inset-0 flex items-center justify-center bg-white/70">
   592→                <span className="text-[11px] font-medium uppercase tracking-widest text-black/40">Sold Out</span>
   593→              </div>
   594→            )}
   595→          </div>
   596→
   597→          {/* Info */}
   598→          <div className="flex flex-col md:pl-12">
   599→            <p className="text-[10px] uppercase tracking-[0.3em] text-black/30">{product.category}</p>
   600→            <h1 className="mt-3 text-display text-[clamp(2rem,5vw,4.5rem)] leading-[0.9] text-black">{product.name}</h1>
   601→            <p className="mt-4 text-2xl font-light">{available ? `$${product.price}.00` : "Sold Out"}</p>
   602→            {enriched.availability === "Low Stock" && <p className="mt-1 text-[11px] text-black/40">Low Stock — only {product.stock} left</p>}
   603→            <p className="mt-4 text-[12px] font-medium uppercase tracking-[0.2em] text-black/40">{product.albumTitle} Collection</p>
   604→            <p className="mt-4 text-sm leading-relaxed text-black/60">{product.description}</p>
   605→
   606→            {/* Size Selector */}
   607→            {enriched.sizes.length > 0 && (
   608→              <div className="mt-8">
   609→                <p className="mb-3 text-[10px] uppercase tracking-[0.2em] text-black/40">Size</p>
   610→                <div className="flex flex-wrap gap-2">
   611→                  {enriched.sizes.map((s) => (
   612→                    <button key={s} onClick={() => setSelectedSize(s)} className={`h-10 min-w-[48px] px-4 text-[12px] uppercase tracking-wider border transition-all cursor-pointer ${selectedSize === s ? "bg-black text-white border-black" : "bg-white text-black/60 border-black/15 hover:border-black/40"}`}>{s}</button>
   613→                  ))}
   614→                </div>
   615→              </div>

   617→
   618→            {/* Add to Cart / Notify */}
   619→            <div className="mt-8">
   620→              {available ? (
   621→                <button onClick={handleAddToCart} className="w-full bg-black py-4 text-[12px] uppercase tracking-[0.2em] text-white hover:bg-black/80 transition-colors cursor-pointer">Add to Cart</button>

   623→                <div className="border border-black/10 py-4 text-center text-[12px] text-black/50">You&apos;ll be notified when available.</div>
   624→              ) : (
   625→                <div className="flex gap-2">
   626→                  <input type="email" placeholder="Email for notification" value={notifyEmail} onChange={(e) => setNotifyEmail(e.target.value)} className="flex-1 border border-black/15 px-4 py-3 text-[12px] placeholder:text-black/25 focus:outline-none focus:border-black/40" />
   627→                  <button onClick={handleNotify} className="bg-black px-6 py-3 text-[12px] uppercase tracking-[0.15em] text-white hover:bg-black/80 transition-colors cursor-pointer">Notify Me</button>
   628→                </div>
   629→              )}
   630→            </div>
   631→
   632→            <div className="mt-6 border-t border-black/5 pt-6">
   633→              <p className="text-[11px] text-black/30">Free shipping on orders over $100. Ships in 3–5 business days. 30-day returns.</p>
   634→            </div>
   635→          </div>
   636→        </div>
   637→
   638→        {/* Related products */}
   639→        {related.length > 0 && (
   640→          <div className="mt-20">
   641→            <p className="mb-8 text-[10px] uppercase tracking-[0.2em] text-black/30">From the same collection</p>
   642→            <div className="grid grid-cols-2 gap-x-4 gap-y-8 sm:grid-cols-3 md:grid-cols-4 lg:gap-x-6">
   643→              {related.map((p) => (
   644→                <ProductCard key={p.slug} product={p} onSelect={() => openRelated(p)} onQuickAdd={() => addToCart()} />
   645→              ))}
   646→            </div>
   647→          </div>
   648→        )}
   649→      </div>
   650→    </section>
   651→  );
