   164→   164→
   165→   165→function ImageGallery({
   166→   166→  total,
   167→   167→  label,
   168→   168→  bgClass = "bg-[#f0f0f0]",
   169→   169→}: {
   170→   170→  total: number;
   171→   171→  label: string;
   172→   172→  bgClass?: string;
   173→   173→}) {
   174→   174→  const [idx, setIdx] = useState(0);
   175→   175→  return (
   176→   176→    <div className={`relative aspect-square overflow-hidden ${bgClass}`}>
   177→   177→      <div className="flex h-full w-full items-center justify-center text-sm text-black/20">
   178→   178→        {label}
   179→   179→      </div>
   180→   180→      {total > 1 && (
   181→   181→        <>
   182→   182→          <button
   183→   183→            onClick={() => setIdx((p) => (p > 0 ? p - 1 : total - 1))}
   184→   184→            className="absolute left-3 top-1/2 -translate-y-1/2 flex h-8 w-8 items-center justify-center bg-white/80 border border-black/5 hover:bg-white transition-colors cursor-pointer"
   185→   185→            aria-label="Previous"
   186→   186→          >
   187→   187→            <ChevronLeft className="h-4 w-4" />
   188→   188→          </button>
   189→   189→          <button
   190→   190→            onClick={() => setIdx((p) => (p < total - 1 ? p + 1 : 0))}
   191→   191→            className="absolute right-3 top-1/2 -translate-y-1/2 flex h-8 w-8 items-center justify-center bg-white/80 border border-black/5 hover:bg-white transition-colors cursor-pointer"
   192→   192→            aria-label="Next"
   193→   193→          >
   194→   194→            <ChevronRight className="h-4 w-4" />
   195→   195→          </button>
   196→   196→          <span className="absolute left-3 bottom-3 text-[11px] text-black/30">
   197→   197→            {idx + 1}/{total}
   198→   198→          </span>
   199→   199→        </>
   200→   200→      )}
   201→   201→    </div>
   202→   202→  );
   203→   203→}
