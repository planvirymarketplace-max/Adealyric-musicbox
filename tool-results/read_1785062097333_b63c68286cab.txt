   239→   239→/* ─── Category Sidebar ─── */
   240→   240→function CategorySidebar({
   241→   241→  selected,
   242→   242→  onChange,
   243→   243→}: {
   244→   244→  selected: ShopCategory[] | null;
   245→   245→  onChange: (cats: ShopCategory[] | null) => void;
   246→   246→}) {
   247→   247→  const [openNodes, setOpenNodes] = useState<Set<string>>(new Set());
   248→   248→
   249→   249→  const toggle = (slug: string) => {
   250→   250→    setOpenNodes((prev) => {
   251→   251→      const next = new Set(prev);
   252→   252→      if (next.has(slug)) next.delete(slug);
   253→   253→      else next.add(slug);
   254→   254→      return next;
   255→   255→    });
   256→   256→  };
   257→   257→
   258→   258→  const handleLeaf = (cats: ShopCategory[]) => {
   259→   259→    onChange(cats);
   260→   260→  };
   261→   261→
   262→   262→  const handleClear = () => {
   263→   263→    onChange(null);
   264→   264→  };
   265→   265→
   266→   266→  function renderNode(node: { label: string; slug: string; categories?: ShopCategory[]; children?: any[] }, depth: number) {
   267→   267→    const hasChildren = node.children && node.children.length > 0;
   268→   268→    const isOpen = openNodes.has(node.slug);
   269→   269→    const isLeaf = !!node.categories;
   270→   270→
   271→   271→    return (
   272→   272→      <div key={node.slug}>
   273→   273→        <button
   274→   274→          onClick={() => {
   275→   275→            if (isLeaf) handleLeaf(node.categories!);
   276→   276→            else if (hasChildren) toggle(node.slug);
   277→   277→          }}
   278→   278→          className={`w-full flex items-center justify-between py-2 text-left text-[12px] tracking-wide cursor-pointer transition-colors hover:text-black ${
   279→   279→            isLeaf && selected && JSON.stringify(selected) === JSON.stringify(node.categories)
   280→   280→              ? "text-black font-medium"
   281→   281→              : "text-black/50"
   282→   282→          }`}
   283→   283→          style={{ paddingLeft: `${depth * 16}px` }}
   284→   284→        >
   285→   285→          <span>{node.label}</span>
   286→   286→          {hasChildren && (
   287→   287→            <ChevronDown className={`h-3 w-3 transition-transform ${isOpen ? "rotate-180" : ""}`} />
   288→   288→          )}
   289→   289→        </button>
   290→   290→        {hasChildren && isOpen && (
   291→   291→          <div>
   292→   292→            {node.children.map((child: any) => renderNode(child, depth + 1))}
   293→   293→          </div>
   294→   294→        )}
   295→   295→      </div>
   296→   296→    );
   297→   297→  }
   298→   298→
   299→   299→  return (
   300→   300→    <div className="w-full">
   301→   301→      <div className="flex items-center justify-between mb-6">
   302→   302→        <p className="text-[10px] uppercase tracking-[0.2em] text-black/30">Categories</p>
   303→   303→        {selected && (
   304→   304→          <button onClick={handleClear} className="text-[10px] uppercase tracking-wider text-black/30 hover:text-black cursor-pointer underline">
   305→   305→            Clear
   306→   306→          </button>
   307→   307→        )}
   308→   308→      </div>
   309→   309→      {CATEGORY_TREE.map((node) => renderNode(node, 0))}
   310→   310→    </div>
   311→   311→  );
   312→   312→}

   314→   314→/* ─── Size Filter (contextual) ─── */
   315→   315→function SizeFilter({
   316→   316→  categories,
   317→   317→  selected,
   318→   318→  onChange,
   319→   319→}: {
   320→   320→  categories: ShopCategory[] | null;
   321→   321→  selected: string | null;
   322→   322→  onChange: (size: string | null) => void;
   323→   323→}) {
   324→   324→  /* Determine which size type applies based on selected categories */
   325→   325→  let activeSizeType: SizeType = "none";
   326→   326→  if (categories) {
   327→   327→    const types = new Set(categories.map((c) => CATEGORY_SIZE_TYPE[c] ?? "none"));
   328→   328→    if (types.size === 1) {
   329→   329→      activeSizeType = [...types][0];
   330→   330→    } else {
   331→   331→      /* Mixed types — default to clothing */
   332→   332→      activeSizeType = "clothing";
   333→   333→    }
   334→   334→  }
   335→   335→
   336→   336→  const options = SIZE_OPTIONS[activeSizeType];
   337→   337→  if (options.length === 0) return null;
   338→   338→
   339→   339→  return (
   340→   340→    <div className="mt-6">
   341→   341→      <p className="text-[10px] uppercase tracking-[0.2em] text-black/30 mb-3">Size</p>
   342→   342→      <div className="flex flex-wrap gap-2">
   343→   343→        {options.map((s) => (
   344→   344→          <button
   345→   345→            key={s}
   346→   346→            onClick={() => onChange(selected === s ? null : s)}
   347→   347→            className={`px-3 py-1.5 text-[11px] uppercase tracking-wider border transition-all cursor-pointer ${
   348→   348→              selected === s
   349→   349→                ? "bg-black text-white border-black"
   350→   350→                : "bg-white text-black/50 border-black/15 hover:border-black/40"
   351→   351→            }`}
   352→   352→          >
   353→   353→            {s}
   354→   354→          </button>
   355→   355→        ))}
   356→   356→      </div>
   357→   357→    </div>
   358→   358→  );
   359→   359→}

   361→   361→/* ─── Shop All View ─── */
   362→   362→function ShopAllView({
   363→   363→  onSelectProduct,
   364→   364→  onQuickAdd,
   365→   365→}: {
   366→   366→  onSelectProduct: (p: ShopProduct) => void;
   367→   367→  onQuickAdd: () => void;
   368→   368→}) {
   369→   369→  const [selectedCats, setSelectedCats] = useState<ShopCategory[] | null>(null);
   370→   370→  const [selectedSize, setSelectedSize] = useState<string | null>(null);
   371→   371→
   372→   372→  const filtered = useMemo(() => {
   373→   373→    let list = ALL_PRODUCTS;
   374→   374→    if (selectedCats) {
   375→   375→      list = list.filter((p) => selectedCats.includes(p.category));
   376→   376→    }
   377→   377→    if (selectedSize) {
   378→   378→      list = list.filter((p) => p.sizes.includes(selectedSize));
   379→   379→    }
   380→   380→    return list;
   381→   381→  }, [selectedCats, selectedSize]);
   382→   382→
   383→   383→  return (
   384→   384→    <div className="mt-8 grid grid-cols-1 gap-12 md:grid-cols-4 lg:grid-cols-5">
   385→   385→      {/* Sidebar */}
   386→   386→      <div className="md:col-span-1">
   387→   387→        <CategorySidebar selected={selectedCats} onChange={setSelectedCats} />
   388→   388→        <SizeFilter categories={selectedCats} selected={selectedSize} onChange={setSelectedSize} />
   389→   389→      </div>
   390→   390→
   391→   391→      {/* Product Grid */}
   392→   392→      <div className="md:col-span-3 lg:col-span-4">
   393→   393→        <p className="mb-4 text-[11px] text-black/30">
   394→   394→          {filtered.length} product{filtered.length !== 1 ? "s" : ""}
   395→   395→        </p>
   396→   396→        <div className="grid grid-cols-2 gap-x-4 gap-y-8 sm:grid-cols-3 lg:gap-x-6">
   397→   397→          {filtered.map((p) => (
   398→   398→            <ProductCard
   399→   399→              key={p.slug}
   400→   400→              product={p}
   401→   401→              onSelect={() => onSelectProduct(p)}
   402→   402→              onQuickAdd={onQuickAdd}
   403→   403→            />
   404→   404→          ))}
   405→   405→        </div>
   406→   406→        {filtered.length === 0 && (
   407→   407→          <div className="mt-20 text-center">
   408→   408→            <p className="text-sm text-black/30">No products match your filters.</p>
   409→   409→          </div>
   410→   410→        )}
   411→   411→      </div>
   412→   412→    </div>
   413→   413→  );
   414→   414→}
