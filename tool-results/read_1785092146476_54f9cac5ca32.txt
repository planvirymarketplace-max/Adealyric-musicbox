   246→/* ─── Category Sidebar ─── */
   247→function CategorySidebar({
   248→  selected,
   249→  onChange,
   250→}: {
   251→  selected: ShopCategory[] | null;
   252→  onChange: (cats: ShopCategory[] | null) => void;
   253→}) {
   254→  const [openNodes, setOpenNodes] = useState<Set<string>>(new Set());
   255→
   256→  const toggle = (slug: string) => {
   257→    setOpenNodes((prev) => {
   258→      const next = new Set(prev);
   259→      if (next.has(slug)) next.delete(slug);
   260→      else next.add(slug);
   261→      return next;
   262→    });
   263→  };
   264→
   265→  const handleLeaf = (cats: ShopCategory[]) => {
   266→    onChange(cats);
   267→  };
   268→
   269→  const handleClear = () => {
   270→    onChange(null);
   271→  };
   272→
   273→  function renderNode(node: { label: string; slug: string; categories?: ShopCategory[]; children?: any[] }, depth: number) {
   274→    const hasChildren = node.children && node.children.length > 0;
   275→    const isOpen = openNodes.has(node.slug);
   276→    const isLeaf = !!node.categories;
   277→
   278→    return (
   279→      <div key={node.slug}>
   280→        <button
   281→          onClick={() => {
   282→            if (isLeaf) handleLeaf(node.categories!);
   283→            else if (hasChildren) toggle(node.slug);
   284→          }}
   285→          className={`w-full flex items-center justify-between py-2 text-left text-[12px] tracking-wide cursor-pointer transition-colors hover:text-black ${
   286→            isLeaf && selected && JSON.stringify(selected) === JSON.stringify(node.categories)
   287→              ? "text-black font-medium"
   288→              : "text-black/50"
   289→          }`}
   290→          style={{ paddingLeft: `${depth * 16}px` }}
   291→        >
   292→          <span>{node.label}</span>
   293→          {hasChildren && (
   294→            <ChevronDown className={`h-3 w-3 transition-transform ${isOpen ? "rotate-180" : ""}`} />
   295→          )}
   296→        </button>
   297→        {hasChildren && isOpen && (
   298→          <div>
   299→            {node.children.map((child: any) => renderNode(child, depth + 1))}
   300→          </div>
   301→        )}
   302→      </div>
   303→    );
   304→  }
   305→
   306→  return (
   307→    <div className="w-full">
   308→      <div className="flex items-center justify-between mb-6">
   309→        <p className="text-[10px] uppercase tracking-[0.2em] text-black/30">Categories</p>
   310→        {selected && (
   311→          <button onClick={handleClear} className="text-[10px] uppercase tracking-wider text-black/30 hover:text-black cursor-pointer underline">
   312→            Clear
   313→          </button>
   314→        )}
   315→      </div>
   316→      {CATEGORY_TREE.map((node) => renderNode(node, 0))}
   317→    </div>
   318→  );
   319→}

   321→/* ─── Size Filter (contextual) ─── */
   322→const SIZE_LABELS: Record<string, string> = {
   323→  clothing: "Clothing Size",
   324→  waist: "Waist Size",
   325→  shoe: "Shoe Size",
   326→  hat: "Hat Size",
   327→  drinkware: "Size",
   328→  "one-size": "",
   329→  none: "",
   330→};
   331→
   332→function SizeFilter({
   333→  categories,
   334→  selected,
   335→  onChange,
   336→}: {
   337→  categories: ShopCategory[] | null;
   338→  selected: string | null;
   339→  onChange: (size: string | null) => void;
   340→}) {
   341→  /* Determine which size type applies based on selected categories */
   342→  let activeSizeType: SizeType = "none";
   343→  if (categories) {
   344→    const types = new Set(categories.map((c) => CATEGORY_SIZE_TYPE[c] ?? "none"));
   345→    if (types.size === 1) {
   346→      activeSizeType = [...types][0];
   347→    } else {
   348→      /* Mixed types — default to clothing */
   349→      activeSizeType = "clothing";
   350→    }
   351→  }
   352→
   353→  const options = SIZE_OPTIONS[activeSizeType];
   354→  const label = SIZE_LABELS[activeSizeType];
   355→  if (!options || options.length === 0 || !label) return null;
   356→
   357→  return (
   358→    <div className="mt-6">
   359→      <p className="text-[10px] uppercase tracking-[0.2em] text-black/30 mb-3">{label}</p>
   360→      <div className="flex flex-wrap gap-2">
   361→        {options.map((s) => (
   362→          <button
   363→            key={s}
   364→            onClick={() => onChange(selected === s ? null : s)}
   365→            className={`px-3 py-1.5 text-[11px] uppercase tracking-wider border transition-all cursor-pointer ${
   366→              selected === s
   367→                ? "bg-black text-white border-black"
   368→                : "bg-white text-black/50 border-black/15 hover:border-black/40"
   369→            }`}
   370→          >
   371→            {s}
   372→          </button>
   373→        ))}
   374→      </div>
   375→    </div>
   376→  );
   377→}

   379→/* ─── Shop All View ─── */
   380→function ShopAllView({
   381→  onSelectProduct,
   382→  onQuickAdd,
   383→}: {
   384→  onSelectProduct: (p: ShopProduct) => void;
   385→  onQuickAdd: () => void;
   386→}) {
   387→  const [selectedCats, setSelectedCats] = useState<ShopCategory[] | null>(null);
   388→  const [selectedSize, setSelectedSize] = useState<string | null>(null);
   389→
   390→  const filtered = useMemo(() => {
   391→    let list = ALL_PRODUCTS;
   392→    if (selectedCats) {
   393→      list = list.filter((p) => selectedCats.includes(p.category));
   394→    }
   395→    if (selectedSize) {
   396→      list = list.filter((p) => p.sizes.includes(selectedSize));
   397→    }
   398→    return list;
   399→  }, [selectedCats, selectedSize]);
   400→
   401→  return (
   402→    <div className="mt-8 grid grid-cols-1 gap-12 md:grid-cols-4 lg:grid-cols-5">
   403→      {/* Sidebar */}
   404→      <div className="md:col-span-1">
   405→        <CategorySidebar selected={selectedCats} onChange={setSelectedCats} />
   406→        <SizeFilter categories={selectedCats} selected={selectedSize} onChange={setSelectedSize} />
   407→      </div>
   408→
   409→      {/* Product Grid */}
   410→      <div className="md:col-span-3 lg:col-span-4">
   411→        <p className="mb-4 text-[11px] text-black/30">
   412→          {filtered.length} product{filtered.length !== 1 ? "s" : ""}
   413→        </p>
   414→        <div className="grid grid-cols-2 gap-x-4 gap-y-8 sm:grid-cols-3 lg:gap-x-6">
   415→          {filtered.map((p) => (
   416→            <ProductCard
   417→              key={p.slug}
   418→              product={p}
   419→              onSelect={() => onSelectProduct(p)}
   420→              onQuickAdd={onQuickAdd}
   421→            />
   422→          ))}
   423→        </div>
   424→        {filtered.length === 0 && (
   425→          <div className="mt-20 text-center">
   426→            <p className="text-sm text-black/30">No products match your filters.</p>
   427→          </div>
   428→        )}
   429→      </div>
   430→    </div>
   431→  );
   432→}

   671→export function ProductDetailPage() {
   672→  const { detailSlug, setDetailSlug, addToCart, setActiveTab } = useAppStore();
   673→  const [selectedSize, setSelectedSize] = useState<string | null>(null);
   674→  const [notifyEmail, setNotifyEmail] = useState("");
   675→  const [notifySent, setNotifySent] = useState(false);
   676→
   677→  const product = useMemo(() => ALL_PRODUCTS.find((p) => p.slug === detailSlug), [detailSlug]);
   678→  const available = product ? product.stock > 0 : false;
   679→  const enriched = product ? enrichProduct(product as ShopProduct) : null;
   680→
   681→  const related = useMemo(() => {
   682→    if (!product) return [];
   683→    return ALL_PRODUCTS.filter((p) => p.albumSlug === product.albumSlug && p.slug !== product.slug).slice(0, 4);
   684→  }, [product]);
   685→
   686→  const openRelated = useCallback((p: EnrichedProduct) => {
   687→    setDetailSlug(p.slug, "product");
   688→    setSelectedSize(null);
   689→    setNotifySent(false);
   690→  }, [setDetailSlug]);
   691→
   692→  const handleAddToCart = useCallback(() => { if (available) addToCart(); }, [available, addToCart]);
   693→  const handleNotify = useCallback(() => { if (notifyEmail) setNotifySent(true); }, [notifyEmail]);
   694→
   695→  if (!product || !enriched) {
   696→    return <div className="flex min-h-screen items-center justify-center bg-white"><p className="text-sm text-black/40">Product not found.</p></div>;
   697→  }
   698→
   699→  return (
   700→    <section className="min-h-screen bg-white px-6 pt-36 pb-20 md:px-12 md:pt-48 md:pb-28">
   701→      <div className="mx-auto max-w-[1400px]">
   702→        <button
   703→          onClick={() => {
   704→            const parentAlbum = SHOP_ALBUMS.find((a) => a.slug === product.albumSlug);
   705→            if (parentAlbum) { setDetailSlug(parentAlbum.slug, "album"); }
   706→            else { setDetailSlug(null, null); setActiveTab("shop"); }
   707→          }}
   708→          className="mb-10 flex items-center gap-2 text-[11px] uppercase tracking-[0.2em] text-black/30 hover:text-black transition-colors cursor-pointer"
   709→        >
   710→          <ArrowLeft className="h-4 w-4" />
   711→          Back
   712→        </button>
   713→
   714→        <div className="grid grid-cols-1 gap-12 md:grid-cols-2 md:gap-16">
   715→          {/* Left: Image */}
   716→          <div>
   717→            <div className="relative aspect-square overflow-hidden bg-[#f0f0f0]">
   718→              {product.image ? (
   719→                <img src={product.image} alt={product.name} className="h-full w-full object-cover" />
   720→              ) : (
   721→                <div className="flex h-full w-full items-center justify-center">
   722→                  <ShoppingBag className="h-12 w-12 text-black/10" />
   723→                </div>
   724→              )}
   725→              {!available && (
   726→                <div className="absolute inset-0 flex items-center justify-center bg-white/70">
   727→                  <span className="text-[11px] font-medium uppercase tracking-widest text-black/40">Sold Out</span>
   728→                </div>
   729→              )}
   730→            </div>
   731→            {product.colors.length > 0 && (
   732→              <div className="mt-4 flex items-center gap-3">
   733→                <span className="text-[10px] uppercase tracking-[0.2em] text-black/30">Color</span>
   734→                {product.colors.map((c) => (
   735→                  <span key={c.hex} className="block h-5 w-5 rounded-full border-2 border-black/10 cursor-pointer hover:border-black/40 transition-colors" style={{ backgroundColor: c.hex }} title={c.name} />
   736→                ))}
   737→              </div>
   738→            )}
   739→          </div>
   740→
   741→          {/* Right: Info */}
   742→          <div className="flex flex-col">
   743→            <p className="text-[10px] uppercase tracking-[0.3em] text-black/30">{product.category}</p>
   744→            <h1 className="mt-3 text-display text-[clamp(2rem,5vw,4.5rem)] leading-[0.9] text-black">{product.name}</h1>
   745→            <p className="mt-4 text-2xl font-light">{available ? `$${product.price}.00` : "Sold Out"}</p>
   746→            {enriched.availability === "Low Stock" && <p className="mt-1 text-[11px] text-black/40">Low Stock — only {product.stock} left</p>}
   747→            <p className="mt-4 text-[12px] font-medium uppercase tracking-[0.2em] text-black/40">{product.albumTitle} Collection</p>
   748→            <p className="mt-4 text-sm leading-relaxed text-black/60">{product.description}</p>
   749→
   750→            {/* Contextual Size Selector */}
   751→            {enriched.sizes.length > 0 && (
   752→              <div className="mt-8">
   753→                <p className="mb-3 text-[10px] uppercase tracking-[0.2em] text-black/40">Size</p>
   754→                <div className="flex flex-wrap gap-2">

   782→            </div>

   788→        </div>

   790→        {/* Related products */}

   792→          <div className="mt-20">
   793→            <p className="mb-8 text-[10px] uppercase tracking-[0.2em] text-black/30">From the same collection</p>

   797→              ))}
   798→            </div>
