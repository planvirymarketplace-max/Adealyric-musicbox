    82→   181→          )}
    83→   182→        </button>
    84→   183→        {hasChildren && isOpen && (
    85→   184→          <div>{node.children.map((child: CategoryNode) => renderNode(child, depth + 1))}</div>
    86→   185→        )}
    87→   186→      </div>
    88→   187→    );
    89→   188→  }
