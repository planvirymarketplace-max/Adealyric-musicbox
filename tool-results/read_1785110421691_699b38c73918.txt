   137→function CategorySidebar({
   138→  selected,
   139→  onChange,
   140→  onNavigate,
   141→}: {
   142→  selected: ShopCategory[] | null;
   143→  onChange: (cats: ShopCategory[] | null) => void;
   144→  onNavigate?: () => void;
   145→}) {
   146→  const [openNodes, setOpenNodes] = useState<Set<string>>(new Set());
   147→
   148→  const toggle = (slug: string) => {
   149→    setOpenNodes((prev) => {
   150→      const next = new Set(prev);
   151→      if (next.has(slug)) next.delete(slug);
   152→      else next.add(slug);
   153→      return next;
   154→    });
   155→  };
   156→
   157→  function renderNode(node: CategoryNode, depth: number) {
   158→    const hasChildren = node.children && node.children.length > 0;
   159→    const isOpen = openNodes.has(node.slug);
   160→    const isLeaf = !!node.categories;
   161→
   162→    return (
   163→      <div key={node.slug}>
   164→        <button
   165→          onClick={() => {
   166→            if (isLeaf) {
   167→              onChange(node.categories!);
   168→              onNavigate?.();
   169→            } else if (hasChildren) toggle(node.slug);
   170→          }}
   171→          className={`w-full flex items-center justify-between py-1.5 text-left text-[11px] tracking-wide cursor-pointer transition-colors hover:text-black ${
   172→            isLeaf && selected && JSON.stringify(selected) === JSON.stringify(node.categories)
   173→              ? "text-black font-medium"
   174→              : "text-black"
   175→          }`}
   176→          style={{ paddingLeft: `${depth * 14}px` }}
   177→        >
   178→          <span>{node.label}</span>
   179→          {hasChildren && (
   180→            <ChevronDown className={`h-3 w-3 transition-transform flex-shrink-0 ${isOpen ? "rotate-180" : ""}`} />
   181→          )}
   182→        </button>
   183→        {hasChildren && isOpen && (
   184→          <div>{node.children.map((child: CategoryNode) => renderNode(child, depth + 1))}</div>
   185→        )}
   186→      </div>
   187→    );
   188→  }
   189→
   190→  return (
   191→    <div className="w-full">
   192→      <div className="flex items-center justify-between mb-4">
   193→        <p className="text-[10px] uppercase tracking-[0.2em] text-black">Categories</p>
   194→        {selected && (
   195→          <button onClick={() => onChange(null)} className="text-[10px] uppercase tracking-wider text-black hover:text-black cursor-pointer underline">Clear</button>
   196→        )}
   197→      </div>
   198→      {CATEGORY_TREE.map((node) => renderNode(node, 0))}
   199→    </div>
   200→  );
   201→}

   544→  const enriched = product ? enrichProduct(product as ShopProduct) : null;
   545→
   546→  const related = useMemo(() => {
   547→    if (!product) return [];
   548→    return ALL_PRODUCTS.filter((p) => p.albumSlug === product.albumSlug && p.slug !== product.slug).slice(0, 4);
   549→  }, [product]);
   550→
   551→  const openRelated = useCallback((p: EnrichedProduct) => {
   552→    setDetailSlug(p.slug, "product");
   553→    setSelectedSize(null);
   554→    setNotifySent(false);
   555→  }, [setDetailSlug]);
   556→
   557→  const handleAddToCart = useCallback(() => { if (available) addToCart(); }, [available, addToCart]);
   558→  const handleNotify = useCallback(() => { if (notifyEmail) setNotifySent(true); }, [notifyEmail]);
   559→
   560→  if (!product || !enriched) {
   561→    return <div className="flex min-h-screen items-center justify-center bg-white"><p className="text-sm text-black">Product not found.</p></div>;
   562→  }
   563→
   564→  return (
   565→    <section className="min-h-screen bg-white px-4 pt-32 pb-20 md:px-8 md:pt-44 md:pb-28">
   566→      <div className="mx-auto max-w-[1400px]">
   567→        {/* Back + nav */}
   568→        <div className="mb-8 flex items-center gap-6">
   569→          <button
   570→            onClick={() => {
   571→              const parentAlbum = SHOP_ALBUMS.find((a) => a.slug === product.albumSlug);
   572→              if (parentAlbum) { setDetailSlug(parentAlbum.slug, "album"); }
   573→              else { setDetailSlug(null, null); setActiveTab("shop"); setShopSubPath(null); }
   574→            }}
   575→            className="flex items-center gap-2 text-[11px] uppercase tracking-[0.2em] text-black hover:text-black transition-colors cursor-pointer"
   576→          >
   577→            <ArrowLeft className="h-4 w-4" />
   578→            Back
   579→          </button>
   580→          <button onClick={() => { setDetailSlug(null, null); setActiveTab("shop"); setShopSubPath("shop-all"); }} className="text-[11px] uppercase tracking-[0.2em] text-black hover:text-black transition-colors cursor-pointer">Shop All</button>
   581→          <button onClick={() => { setDetailSlug(null, null); setActiveTab("shop"); setShopSubPath("collections"); }} className="text-[11px] uppercase tracking-[0.2em] text-black hover:text-black transition-colors cursor-pointer">Collections</button>
   582→        </div>
   583→
   584→        <div className="grid grid-cols-1 gap-8 md:grid-cols-[180px_1fr_1fr] md:gap-0">
   585→          {/* Category Sidebar */}
   586→          <div className="hidden md:block pr-8 border-r border-black/10">
   587→            <CategorySidebar selected={sidebarCats} onChange={setSidebarCats} onNavigate={() => { setDetailSlug(null, null); setActiveTab("shop"); }} />
   588→          </div>
   589→
   590→          {/* Image */}
   591→          <div className="relative aspect-square overflow-hidden bg-[#f4f4f4]">
   592→            {product.image ? (
   593→              <img src={product.image} alt={product.name} className="h-full w-full object-cover" />
   594→            ) : (
   595→              <div className="flex h-full w-full items-center justify-center"><ShoppingBag className="h-12 w-12 text-black/10" /></div>
   596→            )}
   597→            {!available && (
   598→              <div className="absolute inset-0 flex items-center justify-center bg-white/70">
   599→                <span className="text-[11px] font-medium uppercase tracking-widest text-black">Sold Out</span>
   600→              </div>
   601→            )}
   602→          </div>
   603→
   604→          {/* Info */}
   605→          <div className="flex flex-col md:pl-12">
   606→            <p className="text-[10px] uppercase tracking-[0.3em] text-black">{product.category}</p>
   607→            <h1 className="mt-3 text-display text-[clamp(2rem,5vw,4.5rem)] leading-[0.9] text-black">{product.name}</h1>
   608→            <p className="mt-4 text-2xl font-light">{available ? `$${product.price}.00` : "Sold Out"}</p>
   609→            {enriched.availability === "Low Stock" && <p className="mt-1 text-[11px] text-black">Low Stock — only {product.stock} left</p>}
   610→            <p className="mt-4 text-[12px] font-medium uppercase tracking-[0.2em] text-black">{product.albumTitle} Collection</p>
   611→            <p className="mt-4 text-sm leading-relaxed text-black">{product.description}</p>
   612→
   613→            {/* Size Selector */}
   614→            {enriched.sizes.length > 0 && (
   615→              <div className="mt-8">
   616→                <p className="mb-3 text-[10px] uppercase tracking-[0.2em] text-black">Size</p>
   617→                <div className="flex flex-wrap gap-2">
   618→                  {enriched.sizes.map((s) => (
   619→                    <button key={s} onClick={() => setSelectedSize(s)} className={`h-10 min-w-[48px] px-4 text-[12px] uppercase tracking-wider border transition-all cursor-pointer ${selectedSize === s ? "bg-black text-white border-black" : "bg-white text-black border-black/15 hover:border-black/40"}`}>{s}</button>
   620→                  ))}
   621→                </div>
   622→              </div>
   623→            )}
   624→
   625→            {/* Add to Cart / Notify */}
   626→            <div className="mt-8">
   627→              {available ? (
   628→                <button onClick={handleAddToCart} className="w-full bg-black py-4 text-[12px] uppercase tracking-[0.2em] text-white hover:bg-black/80 transition-colors cursor-pointer">Add to Cart</button>
   629→              ) : notifySent ? (
   630→                <div className="border border-black/10 py-4 text-center text-[12px] text-black">You&apos;ll be notified when available.</div>
   631→              ) : (
   632→                <div className="flex gap-2">
   633→                  <input type="email" placeholder="Email for notification" value={notifyEmail} onChange={(e) => setNotifyEmail(e.target.value)} className="flex-1 border border-black/15 px-4 py-3 text-[12px] placeholder:text-black/25 focus:outline-none focus:border-black/40" />
   634→                  <button onClick={handleNotify} className="bg-black px-6 py-3 text-[12px] uppercase tracking-[0.15em] text-white hover:bg-black/80 transition-colors cursor-pointer">Notify Me</button>
   635→                </div>
   636→              )}
   637→            </div>
   638→
   639→            <div className="mt-6 border-t border-black/5 pt-6">
   640→              <p className="text-[11px] text-black">Free shipping on orders over $100. Ships in 3–5 business days. 30-day returns.</p>
   641→            </div>
   642→          </div>
   643→        </div>
   644→
   645→        {/* Related products */}
   646→        {related.length > 0 && (
   647→          <div className="mt-20">
   648→            <p className="mb-8 text-[10px] uppercase tracking-[0.2em] text-black">From the same collection</p>
   649→            <div className="grid grid-cols-2 gap-x-4 gap-y-8 sm:grid-cols-3 md:grid-cols-4 lg:gap-x-6">
   650→              {related.map((p) => (
   651→                <ProductCard key={p.slug} product={p} onSelect={() => openRelated(p)} onQuickAdd={() => addToCart()} />
   652→              ))}
   653→            </div>
   654→          </div>
   655→        )}
   656→      </div>
   657→    </section>
   658→  );
   659→}
