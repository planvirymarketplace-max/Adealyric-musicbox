   687→export type AvailabilityStatus = "In Stock" | "Low Stock" | "Out of Stock" | "Pre-Order";
   688→
   689→function getAvailability(stock: number): AvailabilityStatus {
   690→  if (stock === 0) return "Out of Stock";
   691→  if (stock <= 10) return "Low Stock";
   692→  return "In Stock";
   693→}
