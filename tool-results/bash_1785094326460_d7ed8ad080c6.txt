   428	export type AvailabilityStatus = "In Stock" | "Low Stock" | "Out of Stock" | "Pre-Order";

   430	function getAvailability(stock: number): AvailabilityStatus {
   431	  if (stock === 0) return "Out of Stock";
   432	  if (stock <= 10) return "Low Stock";
   433	  return "In Stock";
   434	}
