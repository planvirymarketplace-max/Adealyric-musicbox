   161→/* ═══════════════════════════════════════════════════
   162→   IMAGE GALLERY
   163→   ═══════════════════════════════════════════════════ */
   164→
   165→function ImageGallery({
   166→  total,
   167→  label,
   168→  bgClass = "bg-[#f0f0f0]",
   169→}: {
   170→  total: number;
   171→  label: string;
   172→  bgClass?: string;
   173→}) {
   174→  const [idx, setIdx] = useState(0);
   175→  return (
   176→    <div className={`relative aspect-square overflow-hidden ${bgClass}`}>
   177→      <div className="flex h-full w-full items-center justify-center text-sm text-black/20">
   178→        {label}
   179→      </div>
   180→      {total > 1 && (
   181→        <>
   182→          <button
   183→            onClick={() => setIdx((p) => (p > 0 ? p - 1 : total - 1))}
   184→            className="absolute left-3 top-1/2 -translate-y-1/2 flex h-8 w-8 items-center justify-center bg-white/80 border border-black/5 hover:bg-white transition-colors cursor-pointer"
   185→            aria-label="Previous"
   186→          >
   187→            <ChevronLeft className="h-4 w-4" />
   188→          </button>
   189→          <button
   190→            onClick={() => setIdx((p) => (p < total - 1 ? p + 1 : 0))}
   191→            className="absolute right-3 top-1/2 -translate-y-1/2 flex h-8 w-8 items-center justify-center bg-white/80 border border-black/5 hover:bg-white transition-colors cursor-pointer"
   192→            aria-label="Next"
   193→          >
   194→            <ChevronRight className="h-4 w-4" />
   195→          </button>
   196→          <span className="absolute left-3 bottom-3 text-[11px] text-black/30">
   197→            {idx + 1}/{total}
   198→          </span>
   199→        </>
   200→      )}
   201→    </div>
   202→  );
   203→}

   491→function ActionButton({
   492→  icon,
   493→  label,
   494→  albumTitle,
   495→}: {
   496→  icon: React.ReactNode;
   497→  label: string;
   498→  albumTitle?: string;
   499→}) {

   507→          onClick={() => setStreamOpen(true)}
   508→          className="group flex w-full items-center gap-3 border-t border-black/10 py-3.5 text-left hover:bg-black/[0.02] transition-colors cursor-pointer"

   518→          open={streamOpen}
   519→          onOpenChange={setStreamOpen}
   520→          title={albumTitle}

   522→      </>
   523→    );

   528→      onClick={() => addToCart()}
   529→      className="group flex w-full items-center gap-3 border-t border-black/10 py-3.5 text-left hover:bg-black/[0.02] transition-colors cursor-pointer"

   538→  );
   539→}
