    46→function useReveal<T extends HTMLElement>() {
    47→  const ref = useRef<T | null>(null);
    48→  const [shown, setShown] = useState(false);
    49→  useEffect(() => {
    50→    if (!ref.current) return;
    51→    const io = new IntersectionObserver(
    52→      ([e]) => e.isIntersecting && setShown(true),
    53→      { threshold: 0.08 }
    54→    );
    55→    io.observe(ref.current);
    56→    return () => io.disconnect();
    57→  }, []);
    58→  return { ref, shown };
    59→}

   696→export function ProductDetailPage() {
   697→  const { detailSlug, setDetailSlug, addToCart } = useAppStore();
   698→  const [imgIdx, setImgIdx] = useState(0);
   699→  const [selectedSize, setSelectedSize] = useState<string | null>(null);
   700→  const [added, setAdded] = useState(false);
   701→  const [notifyEmail, setNotifyEmail] = useState("");
   702→  const [notified, setNotified] = useState(false);
   703→
   704→  // Find product across all albums
   705→  const allProducts = useMemo(() => {
   706→    const products: (AlbumMerchProduct & {
   707→      albumSlug: string;
   708→      albumTitle: string;

   710→    ALBUMS.forEach((album) => {

   719→    return products;
   720→  }, []);
   721→
   722→  const product = allProducts.find((p) => p.slug === detailSlug);
   723→
   724→  // Related: same album, different product
   725→  const related = useMemo(() => {
   726→    if (!product) return [];
   727→    return allProducts

   744→  const totalImgs = 3;
   745→

   751→
   752→  const handleNotify = () => {

   754→      setNotified(true);
   755→      setNotifyEmail("");
   756→    }
   757→  };
   758→
   759→  const goBack = () => {

   761→      setDetailSlug(product.albumSlug, "album");

   765→  };
   766→
   767→  return (

   777→            <ArrowLeft className="h-3.5 w-3.5 transition-transform group-hover:-translate-x-1" />

   781→          <div className="mt-12 grid grid-cols-1 gap-10 md:gap-16 md:grid-cols-2">
   782→            {/* Product image */}

   784→              <div className="relative aspect-square overflow-hidden bg-bone/5">

   786→                  {product.name}
   787→                </div>

   790→                  onClick={() =>

   792→                  }
   793→                  className="absolute left-3 top-1/2 -translate-y-1/2 flex h-8 w-8 items-center justify-center bg-ink/60 border border-bone/10 text-bone/60 hover:text-bone hover:bg-ink/80 transition-colors cursor-pointer"
   794→                  aria-label="Previous image"
   795→                >
   796→                  <ChevronLeft className="h-4 w-4" />
   797→                </button>
   798→                <button
   799→                  onClick={() =>
   800→                    setImgIdx((p) => (p < totalImgs - 1 ? p + 1 : 0))
   801→                  }
   802→                  className="absolute right-3 top-1/2 -translate-y-1/2 flex h-8 w-8 items-center justify-center bg-ink/60 border border-bone/10 text-bone/60 hover:text-bone hover:bg-ink/80 transition-colors cursor-pointer"

   805→                  <ChevronRight className="h-4 w-4" />

   826→                )}
   827→              </p>
