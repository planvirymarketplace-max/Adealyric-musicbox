   339→export type AvailabilityStatus = "In Stock" | "Low Stock" | "Out of Stock" | "Pre-Order";
   340→
   341→function getAvailability(stock: number): AvailabilityStatus {
   342→  if (stock === 0) return "Out of Stock";
   343→  if (stock <= 10) return "Low Stock";
   344→  return "In Stock";
   345→}

   361→export type EnrichedProduct = ShopProduct & {
   362→  sizeType: SizeType;
   363→  availability: AvailabilityStatus;

   366→export function enrichProduct(p: ShopProduct): EnrichedProduct {
   367→  return {
   368→    ...p,
   369→    sizeType: CATEGORY_SIZE_TYPE[p.category] ?? "none",
   370→    availability: getAvailability(p.stock),
   371→  };
