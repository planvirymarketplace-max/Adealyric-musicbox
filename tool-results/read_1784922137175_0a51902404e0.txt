   314→    if (upErr) { setUploading(false); toast('error', 'Upload failed'); return; }
   315→

   321→        audio_storage_path: path,
   322→        duration_seconds: duration,
   323→      }).eq('id', track.id);
   324→      setUploading(false);
   325→      if (error) toast('error', 'Failed to save track info');

   340→      <div className="flex items-start gap-3">

   351→            {!track.download_allowed && <Badge color="gray">No download</Badge>}
   352→          </div>
   353→          <p className="text-xs text-neutral-400 mb-2">
