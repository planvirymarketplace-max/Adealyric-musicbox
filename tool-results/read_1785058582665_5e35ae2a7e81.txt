   638→          </div>
   639→          <div className="grid grid-cols-2 gap-x-4 gap-y-8 sm:grid-cols-3 md:grid-cols-4 lg:gap-x-6">

   686→export function ProductDetailPage() {
   687→  const { detailSlug, setDetailSlug, addToCart, setActiveTab } =
   688→    useAppStore();
   689→  const [selectedSize, setSelectedSize] = useState<string | null>(null);
   690→  const [notifyEmail, setNotifyEmail] = useState("");
   691→  const [notifySent, setNotifySent] = useState(false);
   692→
   693→  const product = useMemo(
   694→    () => ALL_PRODUCTS.find((p) => p.slug === detailSlug),
   695→    [detailSlug]
   696→  );
   697→
   698→  const available = product ? product.stock > 0 : false;
   699→
   700→  /* Related products: same album, different product */
   701→  const related = useMemo(() => {
   702→    if (!product) return [];
   703→    return ALL_PRODUCTS.filter(
   704→      (p) =>
   705→        p.albumSlug === product.albumSlug && p.slug !== product.slug
   706→    ).slice(0, 4);
   707→  }, [product]);
   708→
   709→  const openRelated = useCallback(
   710→    (p: ShopProduct) => {
   711→      setDetailSlug(p.slug, "product");
   712→      setSelectedSize(null);
   713→      setNotifySent(false);
   714→    },
   715→    [setDetailSlug]
   716→  );
   717→
   718→  const handleAddToCart = useCallback(() => {
   719→    if (available) {
   720→      addToCart();
   721→    }
   722→  }, [available, addToCart]);
   723→
   724→  const handleNotify = useCallback(() => {
   725→    if (notifyEmail) {
   726→      setNotifySent(true);
   727→    }
   728→  }, [notifyEmail]);
   729→
   730→  if (!product) {
   731→    return (
   732→      <div className="flex min-h-screen items-center justify-center bg-white">
   733→        <p className="text-sm text-black/40">Product not found.</p>
   734→      </div>
   735→    );
   736→  }
   737→
   738→  return (
   739→    <>
   740→      {/* Dark header */}
   741→      <section className="relative bg-ink px-6 pb-16 pt-36 md:px-12 md:pb-20 md:pt-48">
   742→        <div className="mx-auto max-w-[1400px]">
   743→          <button
   744→            onClick={() => {
   745→              /* Go back to the album page if we came from one, otherwise shop */
   746→              const parentAlbum = SHOP_ALBUMS.find(
   747→                (a) => a.slug === product.albumSlug
   748→              );
   749→              if (parentAlbum) {
   750→                setDetailSlug(parentAlbum.slug, "album");
   751→              } else {
   752→                setDetailSlug(null, null);
   753→                setActiveTab("shop");
   754→              }
   755→            }}
   756→            className="mb-10 flex items-center gap-2 text-[11px] uppercase tracking-[0.2em] text-ash hover:text-bone transition-colors cursor-pointer"
   757→          >
   758→            <ArrowLeft className="h-4 w-4" />
   759→            Back
   760→          </button>
   761→          <p className="text-[10px] uppercase tracking-[0.3em] text-ash">
   762→            {product.category}
   763→          </p>
   764→          <h1 className="mt-4 text-display text-[clamp(2rem,5vw,4.5rem)] leading-[0.9] text-bone">
   765→            {product.name}
   766→          </h1>
   767→        </div>
   768→        <div className="absolute inset-x-0 bottom-0 h-24 bg-gradient-to-t from-white to-transparent" />
   769→      </section>
   770→
   771→      {/* White body */}
   772→      <section className="bg-white px-6 pb-20 md:px-12 md:pb-28">
   773→        <div className="mx-auto max-w-[1400px]">
   774→          <div className="grid grid-cols-1 gap-12 md:grid-cols-2 md:gap-16">
   775→            {/* Left: Image gallery */}
   776→            <div>
   777→              <div className="relative aspect-square overflow-hidden bg-[#f0f0f0]">
   778→                {product.image ? (
   779→                  <img
   780→                    src={product.image}
   781→                    alt={product.name}
   782→                    className="h-full w-full object-cover"
   783→                  />
   784→                ) : (
   785→                  <div className="flex h-full w-full items-center justify-center">
   786→                    <ShoppingBag className="h-12 w-12 text-black/10" />
   787→                  </div>
   788→                )}
   789→                {!available && (
   790→                  <div className="absolute inset-0 flex items-center justify-center bg-white/70">
   791→                    <span className="text-[11px] font-medium uppercase tracking-widest text-black/40">
   792→                      Sold Out
   793→                    </span>
   794→                  </div>
   795→                )}
   796→              </div>
   797→
   798→              {/* Color swatches */}
   799→              {product.colors.length > 0 && (
   800→                <div className="mt-4 flex items-center gap-3">
   801→                  <span className="text-[10px] uppercase tracking-[0.2em] text-black/30">

   808→                      style={{ backgroundColor: c.hex }}
   809→                      title={c.name}
   810→                    />
   811→                  ))}
   812→                </div>
   813→              )}
   814→            </div>
   815→
   816→            {/* Right: Info */}
