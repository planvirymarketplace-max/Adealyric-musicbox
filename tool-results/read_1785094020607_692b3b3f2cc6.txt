   136→   136→
   137→   137→function CategorySidebar({
   138→   138→  selected,
   139→   139→  onChange,
   140→   140→  onNavigate,
   141→   141→}: {
   142→   142→  selected: ShopCategory[] | null;
   143→   143→  onChange: (cats: ShopCategory[] | null) => void;
   144→   144→  onNavigate?: () => void;
   145→   145→}) {
   146→   146→  const [openNodes, setOpenNodes] = useState<Set<string>>(new Set());
   147→   147→
   148→   148→  const toggle = (slug: string) => {
   149→   149→    setOpenNodes((prev) => {
   150→   150→      const next = new Set(prev);
   151→   151→      if (next.has(slug)) next.delete(slug);
   152→   152→      else next.add(slug);
   153→   153→      return next;
   154→   154→    });
   155→   155→  };
   156→   156→
   157→   157→  function renderNode(node: CategoryNode, depth: number) {
   158→   158→    const hasChildren = node.children && node.children.length > 0;
   159→   159→    const isOpen = openNodes.has(node.slug);
   160→   160→    const isLeaf = !!node.categories;
   161→   161→
   162→   162→    return (
   163→   163→      <div key={node.slug}>
   164→   164→        <button
   165→   165→          onClick={() => {
   166→   166→            if (isLeaf) {
   167→   167→              onChange(node.categories!);
   168→   168→              onNavigate?.();
   169→   169→            } else if (hasChildren) toggle(node.slug);
   170→   170→          }}
   171→   171→          className={`w-full flex items-center justify-between py-1.5 text-left text-[11px] tracking-wide cursor-pointer transition-colors hover:text-black ${
   172→   172→            isLeaf && selected && JSON.stringify(selected) === JSON.stringify(node.categories)
   173→   173→              ? "text-black font-medium"
   174→   174→              : "text-black/50"
   175→   175→          }`}
   176→   176→          style={{ paddingLeft: `${depth * 14}px` }}
   177→   177→        >
   178→   178→          <span>{node.label}</span>
   179→   179→          {hasChildren && (
   180→   180→            <ChevronDown className={`h-3 w-3 transition-transform flex-shrink-0 ${isOpen ? "rotate-180" : ""}`} />
   181→   181→          )}
   182→   182→        </button>
   183→   183→        {hasChildren && isOpen && (
   184→   184→          <div>{node.children.map((child: CategoryNode) => renderNode(child, depth + 1))}</div>
   185→   185→        )}
   186→   186→      </div>
   187→   187→    );
   188→   188→  }
   189→   189→
   190→   190→  return (
   191→   191→    <div className="w-full">
   192→   192→      <div className="flex items-center justify-between mb-4">
   193→   193→        <p className="text-[10px] uppercase tracking-[0.2em] text-black/30">Categories</p>
   194→   194→        {selected && (
   195→   195→          <button onClick={() => onChange(null)} className="text-[10px] uppercase tracking-wider text-black/30 hover:text-black cursor-pointer underline">Clear</button>
   196→   196→        )}
   197→   197→      </div>
   198→   198→      {CATEGORY_TREE.map((node) => renderNode(node, 0))}
   199→   199→    </div>
   200→   200→  );
   201→   201→}

   217→   217→function SizeFilter({ categories, selected, onChange }: { categories: ShopCategory[] | null; selected: string | null; onChange: (size: string | null) => void }) {
   218→   218→  let activeSizeType: SizeType = "none";
   219→   219→  if (categories) {
   220→   220→    const types = new Set(categories.map((c) => CATEGORY_SIZE_TYPE[c] ?? "none"));
   221→   221→    if (types.size === 1) activeSizeType = [...types][0];
   222→   222→    else activeSizeType = "clothing";
   223→   223→  }
   224→   224→
   225→   225→  const options = SIZE_OPTIONS[activeSizeType];
   226→   226→  const label = SIZE_LABELS[activeSizeType];
   227→   227→  if (!options || options.length === 0 || !label) return null;
   228→   228→
   229→   229→  return (
   230→   230→    <div className="mt-6">
   231→   231→      <p className="text-[10px] uppercase tracking-[0.2em] text-black/30 mb-3">{label}</p>

   233→   233→        {options.map((s) => (

"bg-black text-white border-black" : "bg-white text-black/50 border-black/15 hover:border-black/40"}`}>{s}</button>
