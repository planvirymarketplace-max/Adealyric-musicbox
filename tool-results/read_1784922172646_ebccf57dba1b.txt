   530→// CSV parser — handles quoted fields, commas inside quotes, newlines inside quotes
   531→function parseCSV(text: string): string[][] {
   532→  const rows: string[][] = [];
   533→  let row: string[] = [];
   534→  let field = '';
   535→  let inQuotes = false;
   536→  let i = 0;
   537→
   538→  while (i < text.length) {
   539→    const char = text[i];
   540→    if (inQuotes) {
   541→      if (char === '"') {
   542→        if (text[i + 1] === '"') { field += '"'; i += 2; continue; }
   543→        inQuotes = false; i++; continue;
   544→      }
   545→      field += char; i++; continue;
   546→    }
   547→    if (char === '"') { inQuotes = true; i++; continue; }
   548→    if (char === ',') { row.push(field); field = ''; i++; continue; }
   549→    if (char === '\r') { i++; continue; }
   550→    if (char === '\n') { row.push(field); rows.push(row); row = []; field = ''; i++; continue; }
   551→    field += char; i++;
   552→  }
   553→  if (field || row.length > 0) { row.push(field); rows.push(row); }
   554→  return rows.filter((r) => r.some((c) => c.trim() !== ''));
   555→}
