   239→/* ─── Category Sidebar ─── */
   240→function CategorySidebar({
   241→  selected,
   242→  onChange,
   243→}: {
   244→  selected: ShopCategory[] | null;
   245→  onChange: (cats: ShopCategory[] | null) => void;
   246→}) {
   247→  const [openNodes, setOpenNodes] = useState<Set<string>>(new Set());
   248→
   249→  const toggle = (slug: string) => {
   250→    setOpenNodes((prev) => {
   251→      const next = new Set(prev);
   252→      if (next.has(slug)) next.delete(slug);
   253→      else next.add(slug);
   254→      return next;
   255→    });
   256→  };
   257→
   258→  const handleLeaf = (cats: ShopCategory[]) => {
   259→    onChange(cats);
   260→  };
   261→
   262→  const handleClear = () => {
   263→    onChange(null);
   264→  };
   265→
   266→  function renderNode(node: { label: string; slug: string; categories?: ShopCategory[]; children?: any[] }, depth: number) {
   267→    const hasChildren = node.children && node.children.length > 0;
   268→    const isOpen = openNodes.has(node.slug);
   269→    const isLeaf = !!node.categories;
   270→
   271→    return (
   272→      <div key={node.slug}>
   273→        <button
   274→          onClick={() => {
   275→            if (isLeaf) handleLeaf(node.categories!);
   276→            else if (hasChildren) toggle(node.slug);
   277→          }}
   278→          className={`w-full flex items-center justify-between py-2 text-left text-[12px] tracking-wide cursor-pointer transition-colors hover:text-black ${
   279→            isLeaf && selected && JSON.stringify(selected) === JSON.stringify(node.categories)
   280→              ? "text-black font-medium"
   281→              : "text-black/50"
   282→          }`}
   283→          style={{ paddingLeft: `${depth * 16}px` }}
   284→        >
   285→          <span>{node.label}</span>
   286→          {hasChildren && (
   287→            <ChevronDown className={`h-3 w-3 transition-transform ${isOpen ? "rotate-180" : ""}`} />
   288→          )}
   289→        </button>
   290→        {hasChildren && isOpen && (
   291→          <div>
   292→            {node.children.map((child: any) => renderNode(child, depth + 1))}
   293→          </div>
   294→        )}
   295→      </div>
   296→    );
   297→  }
   298→
   299→  return (
   300→    <div className="w-full">
   301→      <div className="flex items-center justify-between mb-6">
   302→        <p className="text-[10px] uppercase tracking-[0.2em] text-black/30">Categories</p>
   303→        {selected && (
   304→          <button onClick={handleClear} className="text-[10px] uppercase tracking-wider text-black/30 hover:text-black cursor-pointer underline">
   305→            Clear
   306→          </button>
   307→        )}
   308→      </div>
   309→      {CATEGORY_TREE.map((node) => renderNode(node, 0))}
   310→    </div>
   311→  );
   312→}

   314→/* ─── Size Filter (contextual) ─── */
   315→function SizeFilter({
   316→  categories,
   317→  selected,
   318→  onChange,
   319→}: {
   320→  categories: ShopCategory[] | null;
   321→  selected: string | null;
   322→  onChange: (size: string | null) => void;
   323→}) {
   324→  /* Determine which size type applies based on selected categories */
   325→  let activeSizeType: SizeType = "none";
   326→  if (categories) {
   327→    const types = new Set(categories.map((c) => CATEGORY_SIZE_TYPE[c] ?? "none"));
   328→    if (types.size === 1) {
   329→      activeSizeType = [...types][0];
   330→    } else {
   331→      /* Mixed types — default to clothing */
   332→      activeSizeType = "clothing";
   333→    }
   334→  }
   335→
   336→  const options = SIZE_OPTIONS[activeSizeType];
   337→  if (options.length === 0) return null;
   338→
   339→  return (
   340→    <div className="mt-6">
   341→      <p className="text-[10px] uppercase tracking-[0.2em] text-black/30 mb-3">Size</p>
   342→      <div className="flex flex-wrap gap-2">
   343→        {options.map((s) => (
   344→          <button
   345→            key={s}
   346→            onClick={() => onChange(selected === s ? null : s)}
   347→            className={`px-3 py-1.5 text-[11px] uppercase tracking-wider border transition-all cursor-pointer ${
   348→              selected === s
   349→                ? "bg-black text-white border-black"
   350→                : "bg-white text-black/50 border-black/15 hover:border-black/40"
   351→            }`}
   352→          >
   353→            {s}
   354→          </button>
   355→        ))}
   356→      </div>
   357→    </div>
   358→  );
   359→}

   653→export function ProductDetailPage() {
   654→  const { detailSlug, setDetailSlug, addToCart, setActiveTab } = useAppStore();
   655→  const [selectedSize, setSelectedSize] = useState<string | null>(null);
   656→  const [notifyEmail, setNotifyEmail] = useState("");
   657→  const [notifySent, setNotifySent] = useState(false);
   658→
   659→  const product = useMemo(() => ALL_PRODUCTS.find((p) => p.slug === detailSlug), [detailSlug]);
   660→  const available = product ? product.stock > 0 : false;
   661→  const enriched = product ? enrichProduct(product as ShopProduct) : null;
   662→
   663→  const related = useMemo(() => {
   664→    if (!product) return [];
   665→    return ALL_PRODUCTS.filter((p) => p.albumSlug === product.albumSlug && p.slug !== product.slug).slice(0, 4);
   666→  }, [product]);
   667→
   668→  const openRelated = useCallback((p: EnrichedProduct) => {
   669→    setDetailSlug(p.slug, "product");
   670→    setSelectedSize(null);
   671→    setNotifySent(false);
   672→  }, [setDetailSlug]);
   673→
   674→  const handleAddToCart = useCallback(() => { if (available) addToCart(); }, [available, addToCart]);
   675→  const handleNotify = useCallback(() => { if (notifyEmail) setNotifySent(true); }, [notifyEmail]);
   676→
   677→  if (!product || !enriched) {
   678→    return <div className="flex min-h-screen items-center justify-center bg-white"><p className="text-sm text-black/40">Product not found.</p></div>;
   679→  }
   680→
   681→  return (
   682→    <section className="min-h-screen bg-white px-6 pt-36 pb-20 md:px-12 md:pt-48 md:pb-28">
   683→      <div className="mx-auto max-w-[1400px]">
   684→        <button
   685→          onClick={() => {
   686→            const parentAlbum = SHOP_ALBUMS.find((a) => a.slug === product.albumSlug);
   687→            if (parentAlbum) { setDetailSlug(parentAlbum.slug, "album"); }
   688→            else { setDetailSlug(null, null); setActiveTab("shop"); }
   689→          }}
   690→          className="mb-10 flex items-center gap-2 text-[11px] uppercase tracking-[0.2em] text-black/30 hover:text-black transition-colors cursor-pointer"
   691→        >
   692→          <ArrowLeft className="h-4 w-4" />
   693→          Back
   694→        </button>
   695→
   696→        <div className="grid grid-cols-1 gap-12 md:grid-cols-2 md:gap-16">
   697→          {/* Left: Image */}
   698→          <div>
   699→            <div className="relative aspect-square overflow-hidden bg-[#f0f0f0]">
   700→              {product.image ? (
   701→                <img src={product.image} alt={product.name} className="h-full w-full object-cover" />
   702→              ) : (
   703→                <div className="flex h-full w-full items-center justify-center">
   704→                  <ShoppingBag className="h-12 w-12 text-black/10" />
   705→                </div>
   706→              )}
   707→              {!available && (
   708→                <div className="absolute inset-0 flex items-center justify-center bg-white/70">
   709→                  <span className="text-[11px] font-medium uppercase tracking-widest text-black/40">Sold Out</span>
   710→                </div>
   711→              )}
   712→            </div>
   713→            {product.colors.length > 0 && (
   714→              <div className="mt-4 flex items-center gap-3">
   715→                <span className="text-[10px] uppercase tracking-[0.2em] text-black/30">Color</span>
   716→                {product.colors.map((c) => (
   717→                  <span key={c.hex} className="block h-5 w-5 rounded-full border-2 border-black/10 cursor-pointer hover:border-black/40 transition-colors" style={{ backgroundColor: c.hex }} title={c.name} />
   718→                ))}
   719→              </div>
   720→            )}
   721→          </div>
   722→
   723→          {/* Right: Info */}
   724→          <div className="flex flex-col">
   725→            <p className="text-[10px] uppercase tracking-[0.3em] text-black/30">{product.category}</p>
   726→            <h1 className="mt-3 text-display text-[clamp(2rem,5vw,4.5rem)] leading-[0.9] text-black">{product.name}</h1>
   727→            <p className="mt-4 text-2xl font-light">{available ? `$${product.price}.00` : "Sold Out"}</p>
   728→            {enriched.availability === "Low Stock" && <p className="mt-1 text-[11px] text-black/40">Low Stock — only {product.stock} left</p>}
   729→            <p className="mt-4 text-[12px] font-medium uppercase tracking-[0.2em] text-black/40">{product.albumTitle} Collection</p>
   730→            <p className="mt-4 text-sm leading-relaxed text-black/60">{product.description}</p>
   731→
   732→            {/* Contextual Size Selector */}
   733→            {enriched.sizes.length > 0 && (
   734→              <div className="mt-8">
   735→                <p className="mb-3 text-[10px] uppercase tracking-[0.2em] text-black/40">Size</p>
   736→                <div className="flex flex-wrap gap-2">
   737→                  {enriched.sizes.map((s) => (
   738→                    <button
   739→                      key={s}
   740→                      onClick={() => setSelectedSize(s)}
   741→                      className={`h-10 min-w-[48px] px-4 text-[12px] uppercase tracking-wider border transition-all cursor-pointer ${
   742→                        selectedSize === s ? "bg-black text-white border-black" : "bg-white text-black/60 border-black/15 hover:border-black/40"

   744→                    >{s}</button>

   751→            <div className="mt-8">

   753→                <button onClick={handleAddToCart} className="w-full bg-black py-4 text-[12px] uppercase tracking-[0.2em] text-white hover:bg-black/80 transition-colors cursor-pointer">
   754→                  Add to Cart

   761→                  <button onClick={handleNotify} className="bg-black px-6 py-3 text-[12px] uppercase tracking-[0.15em] text-white hover:bg-black/80 transition-colors cursor-pointer">Notify Me</button>
   762→                </div>
   763→              )}
   764→            </div>
   765→
   766→            <div className="mt-6 border-t border-black/5 pt-6">
   767→              <p className="text-[11px] text-black/30">Free shipping on orders over $100. Ships in 3–5 business days. 30-day returns.</p>
   768→            </div>
   769→          </div>
   770→        </div>
   771→
   772→        {/* Related products */}
   773→        {related.length > 0 && (
   774→          <div className="mt-20">
   775→            <p className="mb-8 text-[10px] uppercase tracking-[0.2em] text-black/30">From the same collection</p>
   776→            <div className="grid grid-cols-2 gap-x-4 gap-y-8 sm:grid-cols-3 md:grid-cols-4 lg:gap-x-6">
   777→              {related.map((p) => (
   778→                <ProductCard key={p.slug} product={p} onSelect={() => openRelated(p)} onQuickAdd={() => addToCart()} />
   779→              ))}
   780→            </div>
   781→          </div>
   782→        )}
   783→      </div>
   784→    </section>
