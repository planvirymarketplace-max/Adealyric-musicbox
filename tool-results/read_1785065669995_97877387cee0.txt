   240→   240→/* ─── Category Sidebar ─── */
   241→   241→function CategorySidebar({
   242→   242→  selected,
   243→   243→  onChange,
   244→   244→}: {
   245→   245→  selected: ShopCategory[] | null;
   246→   246→  onChange: (cats: ShopCategory[] | null) => void;
   247→   247→}) {
   248→   248→  const [openNodes, setOpenNodes] = useState<Set<string>>(new Set());
   249→   249→
   250→   250→  const toggle = (slug: string) => {
   251→   251→    setOpenNodes((prev) => {
   252→   252→      const next = new Set(prev);
   253→   253→      if (next.has(slug)) next.delete(slug);
   254→   254→      else next.add(slug);
   255→   255→      return next;
   256→   256→    });
   257→   257→  };
   258→   258→
   259→   259→  const handleLeaf = (cats: ShopCategory[]) => {
   260→   260→    onChange(cats);
   261→   261→  };
   262→   262→
   263→   263→  const handleClear = () => {
   264→   264→    onChange(null);
   265→   265→  };
   266→   266→
   267→   267→  function renderNode(node: { label: string; slug: string; categories?: ShopCategory[]; children?: any[] }, depth: number) {
   268→   268→    const hasChildren = node.children && node.children.length > 0;
   269→   269→    const isOpen = openNodes.has(node.slug);
   270→   270→    const isLeaf = !!node.categories;
   271→   271→
   272→   272→    return (
   273→   273→      <div key={node.slug}>
   274→   274→        <button
   275→   275→          onClick={() => {
   276→   276→            if (isLeaf) handleLeaf(node.categories!);
   277→   277→            else if (hasChildren) toggle(node.slug);
   278→   278→          }}
   279→   279→          className={`w-full flex items-center justify-between py-2 text-left text-[12px] tracking-wide cursor-pointer transition-colors hover:text-black ${
   280→   280→            isLeaf && selected && JSON.stringify(selected) === JSON.stringify(node.categories)
   281→   281→              ? "text-black font-medium"
   282→   282→              : "text-black/50"
   283→   283→          }`}
   284→   284→          style={{ paddingLeft: `${depth * 16}px` }}
   285→   285→        >
   286→   286→          <span>{node.label}</span>
   287→   287→          {hasChildren && (
   288→   288→            <ChevronDown className={`h-3 w-3 transition-transform ${isOpen ? "rotate-180" : ""}`} />
   289→   289→          )}
   290→   290→        </button>
   291→   291→        {hasChildren && isOpen && (
   292→   292→          <div>
   293→   293→            {node.children.map((child: any) => renderNode(child, depth + 1))}
   294→   294→          </div>
   295→   295→        )}
   296→   296→      </div>
   297→   297→    );
   298→   298→  }
   299→   299→
   300→   300→  return (
   301→   301→    <div className="w-full">
   302→   302→      <div className="flex items-center justify-between mb-6">
   303→   303→        <p className="text-[10px] uppercase tracking-[0.2em] text-black/30">Categories</p>
   304→   304→        {selected && (
   305→   305→          <button onClick={handleClear} className="text-[10px] uppercase tracking-wider text-black/30 hover:text-black cursor-pointer underline">
   306→   306→            Clear
   307→   307→          </button>
   308→   308→        )}
   309→   309→      </div>
   310→   310→      {CATEGORY_TREE.map((node) => renderNode(node, 0))}
   311→   311→    </div>
   312→   312→  );
   313→   313→}

   315→   315→/* ─── Size Filter (contextual) ─── */
   316→   316→const SIZE_LABELS: Record<string, string> = {
   317→   317→  clothing: "Clothing Size",
   318→   318→  waist: "Waist Size",
   319→   319→  shoe: "Shoe Size",
   320→   320→  hat: "Hat Size",
   321→   321→  drinkware: "Size",
   322→   322→  "one-size": "",
   323→   323→  none: "",
   324→   324→};
   325→   325→
   326→   326→function SizeFilter({
   327→   327→  categories,
   328→   328→  selected,
   329→   329→  onChange,
   330→   330→}: {
   331→   331→  categories: ShopCategory[] | null;
   332→   332→  selected: string | null;
   333→   333→  onChange: (size: string | null) => void;
   334→   334→}) {
   335→   335→  /* Determine which size type applies based on selected categories */
   336→   336→  let activeSizeType: SizeType = "none";
   337→   337→  if (categories) {
   338→   338→    const types = new Set(categories.map((c) => CATEGORY_SIZE_TYPE[c] ?? "none"));
   339→   339→    if (types.size === 1) {
   340→   340→      activeSizeType = [...types][0];
   341→   341→    } else {
   342→   342→      /* Mixed types — default to clothing */
   343→   343→      activeSizeType = "clothing";
   344→   344→    }
   345→   345→  }
   346→   346→
   347→   347→  const options = SIZE_OPTIONS[activeSizeType];
   348→   348→  const label = SIZE_LABELS[activeSizeType];
   349→   349→  if (!options || options.length === 0 || !label) return null;
   350→   350→
   351→   351→  return (
   352→   352→    <div className="mt-6">
   353→   353→      <p className="text-[10px] uppercase tracking-[0.2em] text-black/30 mb-3">{label}</p>
   354→   354→      <div className="flex flex-wrap gap-2">
   355→   355→        {options.map((s) => (
   356→   356→          <button
   357→   357→            key={s}
   358→   358→            onClick={() => onChange(selected === s ? null : s)}
   359→   359→            className={`px-3 py-1.5 text-[11px] uppercase tracking-wider border transition-all cursor-pointer ${
   360→   360→              selected === s
   361→   361→                ? "bg-black text-white border-black"
   362→   362→                : "bg-white text-black/50 border-black/15 hover:border-black/40"
   363→   363→            }`}
   364→   364→          >
   365→   365→            {s}
   366→   366→          </button>
   367→   367→        ))}
   368→   368→      </div>
   369→   369→    </div>
   370→   370→  );
   371→   371→}
