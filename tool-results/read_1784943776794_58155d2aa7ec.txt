    46→function useReveal<T extends HTMLElement>() {
    47→  const ref = useRef<T | null>(null);
    48→  const [shown, setShown] = useState(false);
    49→  useEffect(() => {
    50→    if (!ref.current) return;
    51→    const io = new IntersectionObserver(
    52→      ([e]) => e.isIntersecting && setShown(true),
    53→      { threshold: 0.08 }
    54→    );
    55→    io.observe(ref.current);
    56→    return () => io.disconnect();
    57→  }, []);
    58→  return { ref, shown };
    59→}

   697→export function ProductDetailPage() {
   698→  const { detailSlug, setDetailSlug, addToCart } = useAppStore();
   699→  const [imgIdx, setImgIdx] = useState(0);
   700→  const [selectedSize, setSelectedSize] = useState<string | null>(null);
   701→  const [added, setAdded] = useState(false);
   702→  const [notifyEmail, setNotifyEmail] = useState("");
   703→  const [notified, setNotified] = useState(false);
   704→
   705→  // Find product across all albums
   706→  const allProducts = useMemo(() => {
   707→    const products: (AlbumMerchProduct & {
   708→      albumSlug: string;
   709→      albumTitle: string;
   710→    })[] = [];
   711→    ALBUMS.forEach((album) => {
   712→      album.merch.forEach((p) => {
   713→        products.push({
   714→          ...p,
   715→          albumSlug: album.slug,
   716→          albumTitle: album.title,
   717→        });
   718→      });
   719→    });
   720→    return products;
   721→  }, []);
   722→
   723→  const product = allProducts.find((p) => p.slug === detailSlug);
   724→
   725→  // Related: same album, different product
   726→  const related = useMemo(() => {
   727→    if (!product) return [];
   728→    return allProducts
   729→      .filter((p) => p.slug !== product.slug && p.albumSlug === product.albumSlug)
   730→      .slice(0, 4);
   731→  }, [product, allProducts]);
   732→

   734→  useEffect(() => {
   735→    setImgIdx(0);
   736→    setSelectedSize(null);
   737→    setAdded(false);
   738→    setNotifyEmail("");
   739→    setNotified(false);
   740→  }, [detailSlug]);
   741→
   742→  if (!product) return null;
   743→
   744→  const hasSizes = product.sizes && product.sizes.length > 0;
   745→  const totalImgs = 3;
   746→
   747→  const handleAdd = () => {
   748→    setAdded(true);
   749→    addToCart();
   750→    setTimeout(() => setAdded(false), 1500);
   751→  };
   752→
   753→  const handleNotify = () => {
   754→    if (notifyEmail.trim()) {
   755→      setNotified(true);
   756→      setNotifyEmail("");
   757→    }
   758→  };
   759→
   760→  const goBack = () => {
   761→    if (product.albumSlug) {
   762→      setDetailSlug(product.albumSlug, "album");
   763→    } else {
   764→      setDetailSlug(null, null);
   765→    }
   766→  };
   767→
   768→  return (
   769→    <PageShell>
   770→      {/* Dark header */}
   771→      <section className="relative bg-ink grain px-6 pt-36 pb-24 md:px-12 md:pt-44 md:pb-32">
   772→        <div className="grain-overlay" />
   773→        <div className="relative z-10 mx-auto max-w-[1600px]">
   774→          <button
   775→            onClick={goBack}
   776→            className="group flex items-center gap-2 text-eyebrow text-ash hover:text-bone transition-colors cursor-pointer"
   777→          >
   778→            <ArrowLeft className="h-3.5 w-3.5 transition-transform group-hover:-translate-x-1" />
   779→            <span>Back</span>
   780→          </button>
   781→
   782→          <div className="mt-12 grid grid-cols-1 gap-10 md:gap-16 md:grid-cols-2">
   783→            {/* Product image */}
   784→            <div>
   785→              <div className="relative aspect-square overflow-hidden bg-bone/5">
   786→                <div className="flex h-full w-full items-center justify-center text-eyebrow text-bone/20">
   787→                  {product.name}
   788→                </div>
   789→                {/* Nav arrows */}
   790→                <button
   791→                  onClick={() =>
   792→                    setImgIdx((p) => (p > 0 ? p - 1 : totalImgs - 1))
   793→                  }
   794→                  className="absolute left-3 top-1/2 -translate-y-1/2 flex h-8 w-8 items-center justify-center bg-ink/60 border border-bone/10 text-bone/60 hover:text-bone hover:bg-ink/80 transition-colors cursor-pointer"
   795→                  aria-label="Previous image"
   796→                >
   797→                  <ChevronLeft className="h-4 w-4" />
   798→                </button>
   799→                <button
   800→                  onClick={() =>
   801→                    setImgIdx((p) => (p < totalImgs - 1 ? p + 1 : 0))
   802→                  }
   803→                  className="absolute right-3 top-1/2 -translate-y-1/2 flex h-8 w-8 items-center justify-center bg-ink/60 border border-bone/10 text-bone/60 hover:text-bone hover:bg-ink/80 transition-colors cursor-pointer"
   804→                  aria-label="Next image"
   805→                >
   806→                  <ChevronRight className="h-4 w-4" />
   807→                </button>
   808→                {/* Counter */}
   809→                <span className="absolute left-3 bottom-3 text-eyebrow text-bone/30">
   810→                  {imgIdx + 1}/{totalImgs}

   812→              </div>
   813→            </div>

   815→            {/* Product info */}
   816→            <div className="flex flex-col justify-center">
   817→              <div className="text-eyebrow text-ash">{product.category}</div>
   818→              <h1 className="mt-4 font-display text-3xl leading-[0.9] text-bone md:text-5xl">
   819→                {product.name}
   820→              </h1>
   821→
   822→              <p className="mt-4 text-2xl text-bone">
   823→                {product.available ? (
   824→                  `$${product.price}.00`
   825→                ) : (
   826→                  <span className="text-ash">Sold Out</span>
   827→                )}
   828→              </p>
   829→
   830→              <p className="mt-6 text-sm leading-relaxed text-ash/70 max-w-md">
   831→                Official Adea Lyric merchandise. Designed in West Philadelphia.
