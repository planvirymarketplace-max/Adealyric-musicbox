   456→export function ProductDetailPage() {
   457→  const { detailSlug, setDetailSlug, addToCart } = useAppStore();
   458→  const [imgIdx, setImgIdx] = useState(0);
   459→  const [selectedSize, setSelectedSize] = useState<string | null>(null);
   460→  const [added, setAdded] = useState(false);
   461→
   462→  // Find the product across all albums
   463→  const allProducts = useMemo(() => {
   464→    const products: (AlbumMerchProduct & { albumSlug: string; albumTitle: string })[] = [];
   465→    ALBUMS.forEach((album) => {
   466→      album.merch.forEach((p) => {
   467→        products.push({ ...p, albumSlug: album.slug, albumTitle: album.title });
   468→      });
   469→    });
   470→    return products;
   471→  }, []);
   472→
   473→  const product = allProducts.find((p) => p.slug === detailSlug);
   474→
   475→  // Related products: same album, excluding current
   476→  const related = useMemo(() => {
   477→    if (!product) return [];
   478→    return allProducts.filter((p) => p.slug !== product.slug).slice(0, 4);
   479→  }, [product, allProducts]);
   480→
   481→  if (!product) return null;
   482→
   483→  const totalImgs = 3;
   484→  const hasSizes = product.sizes && product.sizes.length > 0;
   485→
   486→  // Auto-select first size when product changes
   487→  if (hasSizes && !selectedSize) {
   488→    const firstAvailable = product.sizes[0];
   489→    setSelectedSize(firstAvailable);
   490→  }
   491→
   492→  const handleAdd = () => {
   493→    setAdded(true);
   494→    addToCart();
   495→    setTimeout(() => setAdded(false), 1500);
   496→  };
   497→
   498→  const goBack = () => {
   499→    // Go back to the album if we came from one
   500→    if (product.albumSlug) {
   501→      setDetailSlug(product.albumSlug, "album");
   502→    } else {
   503→      setDetailSlug(null, null);
   504→    }
   505→  };
   506→
   507→  return (
   508→    <PageShell>
   509→      <div className="min-h-screen bg-white text-[#1A1A1A]">
   510→        {/* Hero */}
   511→        <section className="px-6 pt-28 md:px-12 md:pt-36">
   512→          <div className="max-w-[1600px] mx-auto grid grid-cols-1 md:grid-cols-2 gap-8 md:gap-16">
   513→            {/* Left: Product images */}
   514→            <div>
   515→              <button
   516→                onClick={goBack}
   517→                className="text-xs text-gray-400 hover:text-[#1A1A1A] cursor-pointer uppercase tracking-widest mb-6 inline-block"
   518→              >
   519→                ← Back
   520→              </button>
   521→              <div className="relative aspect-square bg-[#F5F5F5]">
   522→                <PlaceholderImage label={product.name} className="absolute inset-0" />
   523→                {/* Nav arrows */}
   524→                <button
   525→                  onClick={() => setImgIdx((p) => (p > 0 ? p - 1 : totalImgs - 1))}
   526→                  className="absolute left-3 top-1/2 -translate-y-1/2 flex h-8 w-8 items-center justify-center bg-white border border-[#E5E5E5] text-[#1A1A1A] hover:bg-[#F5F5F5] cursor-pointer"
   527→                  aria-label="Previous image"
   528→                >
   529→                  <ChevronLeft className="h-4 w-4" />
