   240→/* ─── Category Sidebar ─── */
   241→function CategorySidebar({
   242→  selected,
   243→  onChange,
   244→}: {
   245→  selected: ShopCategory[] | null;
   246→  onChange: (cats: ShopCategory[] | null) => void;
   247→}) {
   248→  const [openNodes, setOpenNodes] = useState<Set<string>>(new Set());
   249→
   250→  const toggle = (slug: string) => {
   251→    setOpenNodes((prev) => {
   252→      const next = new Set(prev);
   253→      if (next.has(slug)) next.delete(slug);
   254→      else next.add(slug);
   255→      return next;
   256→    });
   257→  };
   258→
   259→  const handleLeaf = (cats: ShopCategory[]) => {
   260→    onChange(cats);
   261→  };
   262→
   263→  const handleClear = () => {
   264→    onChange(null);
   265→  };
   266→
   267→  function renderNode(node: { label: string; slug: string; categories?: ShopCategory[]; children?: any[] }, depth: number) {
   268→    const hasChildren = node.children && node.children.length > 0;
   269→    const isOpen = openNodes.has(node.slug);
   270→    const isLeaf = !!node.categories;
   271→
   272→    return (
   273→      <div key={node.slug}>
   274→        <button
   275→          onClick={() => {
   276→            if (isLeaf) handleLeaf(node.categories!);
   277→            else if (hasChildren) toggle(node.slug);
   278→          }}
   279→          className={`w-full flex items-center justify-between py-2 text-left text-[12px] tracking-wide cursor-pointer transition-colors hover:text-black ${
   280→            isLeaf && selected && JSON.stringify(selected) === JSON.stringify(node.categories)
   281→              ? "text-black font-medium"
   282→              : "text-black/50"
   283→          }`}
   284→          style={{ paddingLeft: `${depth * 16}px` }}
   285→        >
   286→          <span>{node.label}</span>
   287→          {hasChildren && (
   288→            <ChevronDown className={`h-3 w-3 transition-transform ${isOpen ? "rotate-180" : ""}`} />
   289→          )}
   290→        </button>
   291→        {hasChildren && isOpen && (
   292→          <div>
   293→            {node.children.map((child: any) => renderNode(child, depth + 1))}
   294→          </div>
   295→        )}
   296→      </div>
   297→    );
   298→  }
   299→
   300→  return (
   301→    <div className="w-full">
   302→      <div className="flex items-center justify-between mb-6">
   303→        <p className="text-[10px] uppercase tracking-[0.2em] text-black/30">Categories</p>
   304→        {selected && (
   305→          <button onClick={handleClear} className="text-[10px] uppercase tracking-wider text-black/30 hover:text-black cursor-pointer underline">
   306→            Clear
   307→          </button>
   308→        )}
   309→      </div>
   310→      {CATEGORY_TREE.map((node) => renderNode(node, 0))}
   311→    </div>
   312→  );
   313→}

   315→/* ─── Size Filter (contextual) ─── */
   316→const SIZE_LABELS: Record<string, string> = {
   317→  clothing: "Clothing Size",
   318→  waist: "Waist Size",
   319→  shoe: "Shoe Size",
   320→  hat: "Hat Size",
   321→  drinkware: "Size",
   322→  "one-size": "",
   323→  none: "",
   324→};
   325→
   326→function SizeFilter({
   327→  categories,
   328→  selected,
   329→  onChange,
   330→}: {
   331→  categories: ShopCategory[] | null;
   332→  selected: string | null;
   333→  onChange: (size: string | null) => void;
   334→}) {
   335→  /* Determine which size type applies based on selected categories */
   336→  let activeSizeType: SizeType = "none";
   337→  if (categories) {
   338→    const types = new Set(categories.map((c) => CATEGORY_SIZE_TYPE[c] ?? "none"));
   339→    if (types.size === 1) {
   340→      activeSizeType = [...types][0];
   341→    } else {
   342→      /* Mixed types — default to clothing */
   343→      activeSizeType = "clothing";
   344→    }
   345→  }
   346→
   347→  const options = SIZE_OPTIONS[activeSizeType];
   348→  const label = SIZE_LABELS[activeSizeType];
   349→  if (!options || options.length === 0 || !label) return null;
   350→
   351→  return (
   352→    <div className="mt-6">
   353→      <p className="text-[10px] uppercase tracking-[0.2em] text-black/30 mb-3">{label}</p>
   354→      <div className="flex flex-wrap gap-2">
   355→        {options.map((s) => (
   356→          <button
   357→            key={s}
   358→            onClick={() => onChange(selected === s ? null : s)}
   359→            className={`px-3 py-1.5 text-[11px] uppercase tracking-wider border transition-all cursor-pointer ${
   360→              selected === s
   361→                ? "bg-black text-white border-black"
   362→                : "bg-white text-black/50 border-black/15 hover:border-black/40"
   363→            }`}
   364→          >
   365→            {s}
   366→          </button>
   367→        ))}
   368→      </div>
   369→    </div>
   370→  );
   371→}

   373→/* ─── Shop All View ─── */
   374→function ShopAllView({
   375→  onSelectProduct,
   376→  onQuickAdd,
   377→}: {
   378→  onSelectProduct: (p: ShopProduct) => void;
   379→  onQuickAdd: () => void;
   380→}) {
   381→  const [selectedCats, setSelectedCats] = useState<ShopCategory[] | null>(null);
   382→  const [selectedSize, setSelectedSize] = useState<string | null>(null);
   383→
   384→  const filtered = useMemo(() => {
   385→    let list = ALL_PRODUCTS;
   386→    if (selectedCats) {
   387→      list = list.filter((p) => selectedCats.includes(p.category));
   388→    }
   389→    if (selectedSize) {
   390→      list = list.filter((p) => p.sizes.includes(selectedSize));
   391→    }
   392→    return list;
   393→  }, [selectedCats, selectedSize]);
   394→
   395→  return (
   396→    <div className="mt-8 grid grid-cols-1 gap-12 md:grid-cols-4 lg:grid-cols-5">
   397→      {/* Sidebar */}
   398→      <div className="md:col-span-1">
   399→        <CategorySidebar selected={selectedCats} onChange={setSelectedCats} />
   400→        <SizeFilter categories={selectedCats} selected={selectedSize} onChange={setSelectedSize} />
   401→      </div>
   402→
   403→      {/* Product Grid */}
   404→      <div className="md:col-span-3 lg:col-span-4">
   405→        <p className="mb-4 text-[11px] text-black/30">
   406→          {filtered.length} product{filtered.length !== 1 ? "s" : ""}
   407→        </p>
   408→        <div className="grid grid-cols-2 gap-x-4 gap-y-8 sm:grid-cols-3 lg:gap-x-6">
   409→          {filtered.map((p) => (
   410→            <ProductCard
   411→              key={p.slug}
   412→              product={p}
   413→              onSelect={() => onSelectProduct(p)}
   414→              onQuickAdd={onQuickAdd}
   415→            />
   416→          ))}
   417→        </div>
   418→        {filtered.length === 0 && (
   419→          <div className="mt-20 text-center">
   420→            <p className="text-sm text-black/30">No products match your filters.</p>
   421→          </div>
   422→        )}
   423→      </div>
   424→    </div>
   425→  );
   426→}
